% Purpose :
%   To display two Leibnizian routes associated with the Identity of
%   Indiscernibles and to mark the further question of ontological reduction
%   as one that requires independent argument.
%
% Representational limit :
%   The dashed arrow marks an inference that requires additional argument; it
%   is not presented as a conclusion established by the principle itself.
%
% Dependencies :
%   TikZ; libraries used: arrows.meta, calc, positioning. All are loaded by
%   the project preamble; no per-file package or library loading.
%
% =============================================================================

\begin{figure}[htbp]
    \centering
    \hyphenpenalty=10000 \exhyphenpenalty=10000\relax  % no hyphenation inside diagram labels

    \begin{tikzpicture}[
        font=\small,
        node distance=1.1cm,
        % --- base box ---
        box/.style={
            draw, rounded corners=3pt, align=center,
            inner sep=6pt, line width=0.7pt
        },
        % --- node roles ---
        premise/.style={box, text width=4.8cm, minimum height=1.9cm,
                        fill=blue!8, draw=blue!55!black},
        principle/.style={box, text width=6.0cm, minimum height=1.5cm,
                          fill=blue!16, draw=blue!55!black},
        openquestion/.style={box, text width=10.6cm, minimum height=1.5cm,
                             dashed, fill=gray!10, draw=black!60},
        % --- arrows ---
        directed/.style={-{Stealth[length=2.5mm]}, line width=0.9pt},
        optional/.style={-{Stealth[length=2.5mm]}, line width=0.9pt, dashed}
    ]

    % ---------------------------------------------------------------
    % Two Leibnizian premises
    % ---------------------------------------------------------------
    \node[premise] (cic) {
        \textbf{Complete Individual Concept}\\[2pt]
        Every true predicate belongs to the complete notion of the individual
    };
    \node[premise, right=0.8cm of cic] (psr) {
        \textbf{Principle of Sufficient Reason}\\[2pt]
        There must be a sufficient reason why things are as they are
    };

    % ---------------------------------------------------------------
    % Principle reached by each route
    % ---------------------------------------------------------------
    \coordinate (mid) at ($(cic.south)!0.5!(psr.south)$);
    \node[principle, below=1.2cm of mid] (pii) {
        \textbf{Identity of Indiscernibles}\\[2pt]
        No two distinct substances are wholly indiscernible
    };

    % ---------------------------------------------------------------
    % Further ontological question
    % ---------------------------------------------------------------
    \node[openquestion, below=1.3cm of pii] (gap) {
        \textbf{Further ontological question}\\[2pt]
        Does the completeness of an individual's predicates establish that
        predicate-totality is identical with the individual's ontological
        Identity?
    };

    \draw[directed] (cic.south) -- (pii.north west);
    \draw[directed] (psr.south) -- (pii.north east);
    \draw[optional] (pii) -- (gap);

    % ---------------------------------------------------------------
    % Legend
    % ---------------------------------------------------------------
    \node[below=0.6cm of gap, font=\scriptsize] {
        \tikz[baseline=-0.5ex]\draw[directed] (0,0) -- (0.7,0);~route to the principle
        \quad\quad
        \tikz[baseline=-0.5ex]\draw[optional] (0,0) -- (0.7,0);~inference requiring additional argument
    };

    \end{tikzpicture}

    \caption{Two Leibnizian routes associated with the Identity of
    Indiscernibles and the further question of ontological
    reduction. The dashed arrow marks an inference that requires
    additional argument; it is not presented as a conclusion
    established by the principle itself.}
    \label{fig:leibniz-routes-to-indiscernibles}
\end{figure}
