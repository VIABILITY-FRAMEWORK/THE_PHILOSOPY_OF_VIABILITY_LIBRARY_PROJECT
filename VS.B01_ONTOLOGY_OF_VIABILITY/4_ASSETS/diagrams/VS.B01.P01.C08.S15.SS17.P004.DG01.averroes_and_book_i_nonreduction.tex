%   Being and Identity in Book I.
%
% Philosophical Limits :
%   1. The upper structure is a schematic representation of
%      Averroes's account of sensible substance.
%   2. The lower structure is not a composition of two entities.
%   3. The relation between Being and Identity is not represented
%      as a causal or temporal sequence.
%   4. Matter, form, essence, unity, individuation, persistence,
%      and manifestation are not presented as components of Identity.
%   5. The diagram does not claim that Averroes formulated the
%      Book I distinction.

\begin{figure}[htbp]
\centering

\begin{tikzpicture}[
    font=\small,
    every node/.style={inner sep=7pt},
    matternode/.style={
        draw=ViabilityBlue, fill=ViabilityPaleBlue,
        rounded corners=2pt, line width=0.8pt,
        text width=4.6cm, align=center, minimum height=1.7cm
    },
    formnode/.style={
        draw=ViabilityGold, fill=ViabilityPaleGold,
        rounded corners=2pt, line width=0.8pt,
        text width=4.6cm, align=center, minimum height=1.7cm
    },
    composite/.style={
        draw=ViabilityGreen, fill=ViabilityPaleGreen,
        rounded corners=2pt, line width=0.9pt,
        text width=10.8cm, align=center, minimum height=1.5cm
    },
    beingnode/.style={
        draw=ViabilityDarkBlue, fill=ViabilityPaleBlue,
        rounded corners=2pt, line width=0.9pt,
        text width=3.2cm, align=center, minimum height=1.2cm
    },
    identitynode/.style={
        draw=ViabilityGold, fill=ViabilityPaleGold,
        rounded corners=2pt, line width=0.9pt,
        text width=3.2cm, align=center, minimum height=1.2cm
    },
    note/.style={
        draw=gray, fill=white, rounded corners=2pt,
        text width=11cm, align=center
    },
    relation/.style={
        -{Stealth[length=2mm]},
        draw=ViabilityDarkBlue, line width=0.8pt
    },
    nondirectional/.style={
        draw=ViabilityDarkBlue, line width=1pt
    },
    boundary/.style={
        densely dashed, draw=gray, line width=0.75pt
    },
    title/.style={
        font=\small\bfseries, align=center, text width=12cm
    },
    label/.style={
        font=\scriptsize, inner sep=2pt, fill=white
    }
]

% ---------------------------------------------------------------------
% PANEL A — AVERROIST EXPLANATORY ARCHITECTURE
% ---------------------------------------------------------------------

\node[title] (titleA) at (0,0)
    {Averroes: The Constitution of Sensible Substance};

\node[matternode, below=6mm of titleA, xshift=-3.0cm] (matter) {
    \textbf{Prime Matter}\\[3pt]
    Potential subject for substantial forms
};

\node[formnode, below=6mm of titleA, xshift=3.0cm] (form) {
    \textbf{Substantial Form}\\[3pt]
    Actuality and determinate nature
};

\node[composite, below=1.8cm of titleA |- matter.south] (substance) {
    \textbf{Composite Sensible Substance}\\[4pt]
    A determinate existent constituted through matter and form
};

\node[composite, below=1.3cm of substance] (accidents) {
    \textbf{Accidental Determinations}\\[3pt]
    Dependent features that may be gained or lost while
    the substance remains their subject
};

% Vertical arrows into the composite

\draw[relation] (matter.south) -- (matter.south |- substance.north)
    node[pos=0.5, left, label, align=right] {material\\potentiality};

\draw[relation] (form.south) -- (form.south |- substance.north)
    node[pos=0.5, right, label, align=left] {actualization and\\determination};

\draw[relation] (substance) -- (accidents)
    node[pos=0.5, right, label] {subject and dependence};

% ---------------------------------------------------------------------
% PANEL B — BOOK I FOUNDATIONAL RELATION
% ---------------------------------------------------------------------

\node[title, below=1.5cm of accidents] (titleB)
    {Book I: The Relation between Being and Identity};

\node[beingnode, below=5mm of titleB, xshift=-4.2cm] (being)
    {\textbf{Being}};

\node[identitynode, below=5mm of titleB, xshift=4.2cm] (identity)
    {\textbf{Identity}};

\draw[nondirectional] (being.east) -- (identity.west)
    node[pos=0.5, label, text width=3.3cm, align=center,
         text=ViabilityDarkBlue] {
        Analytical Distinction\\
        Ontological Inseparability\\
        Absence of Ontological Priority
    };

\coordinate (axisB) at (titleB |- being.south);

\node[note, below=1.2cm of axisB] (nonreduction) {
    \textbf{Non-equivalence}\\[3pt]
    Matter \quad Form \quad Essence \quad Unity \quad
    Individuation \quad Persistence \quad Manifestation\\[3pt]
    These concepts perform distinct explanatory functions;
    none is identified with Identity.
};

% Dashed boundaries: vertical drops, no crossing

\draw[boundary] (being.south) -- (being.south |- nonreduction.north);
\draw[boundary] (identity.south) -- (identity.south |- nonreduction.north);

\end{tikzpicture}

\caption[
    Averroes's account of substance and the Book I relation
    between Being and Identity
]{
    Schematic comparison between Averroes's account of
    sensible substance through matter and substantial form
    and the foundational relation between Being and Identity
    in Book I. The diagram distinguishes the explanatory
    roles of matter, form, essence, and accidental determination
    from the analytical distinction and ontological inseparability
    of Being and Identity. The two panels represent different
    explanatory questions: the lower relation is not a material
    or formal composition, and the dashed boundaries do not
    indicate causal dependence.
}

\label{fig:VS.B01.P01.C08.S15.SS17.P004.DG01}
