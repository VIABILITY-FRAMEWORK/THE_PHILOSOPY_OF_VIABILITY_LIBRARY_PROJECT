% Purpose :
%   To show ontological dependence and return in the Plotinian structure
%   without temporalizing it.
%
% Representational limit :
%   The diagram represents asymmetrical ontological dependence and return,
%   not a temporal chain of events.
%
% Dependencies :
%   TikZ; libraries used: arrows.meta, positioning. All are loaded by the
%   project preamble; no per-file package or library loading.
%
% =============================================================================

\begin{figure}[htbp]
    \centering
    \hyphenpenalty=10000 \exhyphenpenalty=10000\relax  % no hyphenation inside diagram labels

    \begin{tikzpicture}[
        font=\small,
        node distance=1.5cm,
        % --- base box ---
        box/.style={
            draw, rounded corners=3pt, align=center,
            text width=5.2cm, minimum height=1.4cm,
            inner sep=5pt, line width=0.7pt, fill=blue!8,
            draw=blue!55!black
        },
        % --- arrows and labels ---
        directed/.style={-{Stealth[length=2.5mm]}, line width=0.9pt},
        returnarrow/.style={-{Stealth[length=2.5mm]}, line width=0.9pt,
                            dashed, draw=green!40!black},
        elabel/.style={font=\footnotesize\itshape, text=black!75,
                       align=left, text width=3.0cm}
    ]

    % ---------------------------------------------------------------
    % Four levels
    % ---------------------------------------------------------------
    \node[box] (one) {
        \textbf{The One / Good}\\[2pt] Absolute simplicity
    };
    \node[box, below=of one] (nous) {
        \textbf{Intellect}\\[2pt] Being and intelligible multiplicity
    };
    \node[box, below=of nous] (soul) {
        \textbf{Soul}\\[2pt] Life and mediation
    };
    \node[box, below=of soul] (cosmos) {
        \textbf{Sensible Cosmos}\\[2pt] Ordered, differentiated manifestation
    };

    % ---------------------------------------------------------------
    % Procession (solid, downward; labels to the left of the axis)
    % ---------------------------------------------------------------
    \draw[directed] (one)  -- node[elabel, left=6pt, align=right]
        {\textit{Procession}\\ ontological dependence} (nous);
    \draw[directed] (nous) -- node[elabel, left=6pt, align=right]
        {\textit{Procession}\\ derivative activity} (soul);
    \draw[directed] (soul) -- node[elabel, left=6pt, align=right]
        {\textit{Productive activity}\\ formative order} (cosmos);

    % ---------------------------------------------------------------
    % Return (dashed, upward; labels to the right)
    % ---------------------------------------------------------------
    \draw[returnarrow] (nous.east) .. controls +(1.8,0.2) and +(1.8,-0.2) ..
        node[elabel, right, xshift=0.6cm]
        {\textit{Reversion}\\ toward the source} (one.east);
    \draw[returnarrow] (soul.east) .. controls +(2.2,0.2) and +(2.2,-0.2) ..
        node[elabel, right, xshift=1.0cm]
        {\textit{Contemplative orientation}} (nous.east);

    % ---------------------------------------------------------------
    % Legend and note
    % ---------------------------------------------------------------
    \node[below=0.7cm of cosmos, font=\scriptsize] (legend) {
        \tikz[baseline=-0.5ex]\draw[directed] (0,0) -- (0.7,0);~procession
        \quad\quad
        \tikz[baseline=-0.5ex]\draw[returnarrow] (0,0) -- (0.7,0);~reversion
    };
    \node[below=0.3cm of legend, text width=10cm, align=center,
          font=\footnotesize\itshape] {
        Schematic representation only: the relations are ontological,
        not temporal stages.
    };

    \end{tikzpicture}

    \caption{The Plotinian structure of procession and reversion. The
    diagram represents asymmetrical ontological dependence and
    return, not a temporal chain of events.}
    \label{fig:plotinian-procession-reversion}
\end{figure}
