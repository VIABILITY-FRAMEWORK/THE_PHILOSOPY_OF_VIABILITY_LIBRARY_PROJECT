\begin{table}[htbp]
    \centering
    \small
    \renewcommand{\arraystretch}{1.35}
    \setlength{\tabcolsep}{5pt}

    \caption{What an invariance finding establishes and what
    it does not establish by itself}
    \label{tab:inferential-limits-of-invariance}

    \begin{tabular}{
        |p{0.22\textwidth}
        |p{0.32\textwidth}
        |p{0.32\textwidth}|
    }
        \hline

        \textbf{Analytical Finding}
        &
        \textbf{Directly Warranted}
        &
        \textbf{Not Established by the Finding Alone}
        \\ \hline

        A determination is preserved under a specified
        transformation
        &
        Preservation of that determination within the
        stated domain, respect, criterion, and conditions
        &
        Numerical identity, persistence, continuity,
        or total preservation
        \\ \hline

        The same determination occurs in distinct existents
        &
        A specified correspondence or shared feature
        &
        Numerical identity or ontological unity of
        those existents
        \\ \hline

        A determination changes under a transformation
        &
        Alteration of that determination within the
        examined respect
        &
        Loss of identity, termination of persistence,
        or discontinuity
        \\ \hline

        No invariant has been identified
        &
        No invariance has been established by the
        analysis performed
        &
        The ontological absence of every invariant,
        identity, or organization
        \\ \hline

        A formal invariant is established
        &
        Preservation within the relevant formal system
        and its specified rules
        &
        The ontological instantiation of the formal
        relation without further interpretation
        \\ \hline

        Structural or compositional invariance is established
        &
        Preservation of the specified organizational
        or compositional determination
        &
        Numerical identity or exhaustive preservation
        of the existent
        \\ \hline

        A correspondence is identified across domains
        &
        A specified relation or correspondence between
        the examined determinations
        &
        Ontological identity or automatic transfer
        of invariance between domains
        \\ \hline

    \end{tabular}
\end{table}
