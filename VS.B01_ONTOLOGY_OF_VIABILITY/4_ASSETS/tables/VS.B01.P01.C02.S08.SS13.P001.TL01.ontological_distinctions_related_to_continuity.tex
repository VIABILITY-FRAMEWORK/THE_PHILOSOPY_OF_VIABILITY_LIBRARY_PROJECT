\begin{table}[htbp]
    \centering
    \small
    \renewcommand{\arraystretch}{1.35}
    \setlength{\tabcolsep}{5pt}

    \begin{tabular}{|p{2.7cm}|p{5.0cm}|p{5.0cm}|}
        \hline
        \textbf{Concept} &
        \textbf{Primary Question} &
        \textbf{What It Does Not Establish by Itself}
        \\
        \hline

        Invariance &
        Is a specified determination preserved under
        a specified transformation and criterion? &
        Continuity, persistence, or numerical identity.
        \\
        \hline

        Continuity &
        What determinate relation connects successive
        states, phases, or stages? &
        The numerical identity or persistence of one
        existent.
        \\
        \hline

        Persistence &
        Does an existent continue to exist across a
        specified temporal or transformational domain? &
        A universal model of continuity or a particular
        invariant.
        \\
        \hline

        Numerical Identity &
        Is the existent under consideration one and
        the same existent? &
        The particular relation through which successive
        states or phases are connected.
        \\
        \hline

        Stability &
        Is a condition, configuration, or mode of
        organization maintained under specified
        circumstances? &
        Continuity, persistence, or numerical identity.
        \\
        \hline
    \end{tabular}

    \caption{Distinct analytical questions associated with
    invariance, continuity, persistence, numerical identity,
    and stability. The distinctions do not imply that these
    concepts are unrelated; they prevent one from being used
    as an unqualified substitute for another.}
    \label{tab:ontological-distinctions-related-to-continuity}
\end{table}
