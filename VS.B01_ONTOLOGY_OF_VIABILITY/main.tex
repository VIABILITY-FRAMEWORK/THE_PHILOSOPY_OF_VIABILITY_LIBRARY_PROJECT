% The front cover is always the first page and the back cover is

% pages. Cross-references via the TikZ page overlay need two runs.

\usepackage{graphicx}
\usepackage{tikz}

\newcommand{\FrontCoverFile}{4_ASSETS/cover/front_cover.png}
\newcommand{\BackCoverFile}{4_ASSETS/cover/back_cover.png}

% \CoverPage{<image file>}: one empty, unnumbered page filled edge to edge
\newcommand{\CoverPage}[1]{%
    \IfFileExists{#1}{}{\PackageError{main}{Cover image not found: #1}{Check the path.}}%
    \clearpage
    \thispagestyle{empty}%
    \begin{tikzpicture}[remember picture, overlay]
        \node[inner sep=0pt, outer sep=0pt] at (current page.center)
            {\includegraphics[width=\paperwidth, height=\paperheight]{#1}};
    \end{tikzpicture}%
    \null
    \clearpage
}
